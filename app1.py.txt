# =====================================================
# DATABASE
# =====================================================

def init_db():

    conn = sqlite3.connect(DATABASE)

    cursor = conn.cursor()

    cursor.execute("""
        CREATE TABLE IF NOT EXISTS problems (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            name TEXT NOT NULL,
            enrollment TEXT NOT NULL,
            description TEXT NOT NULL,
            solution TEXT,
            status TEXT DEFAULT 'Pending',
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        )
    """)

    conn.commit()
    conn.close()


# =====================================================
# MENTOR LOGIN PROTECTION
# =====================================================

def mentor_required(function):

    @wraps(function)
    def wrapper(*args, **kwargs):

        if not session.get("mentor_logged_in"):
            return redirect(url_for("mentor_login"))

        return function(*args, **kwargs)

    return wrapper


# =====================================================
# SEND EMAIL TO MENTOR
# =====================================================

def send_email(name, enrollment, description):

    subject = "New Student Problem - GUNI Trouble-Shooter"

    body = f"""
GUNI TROUBLE-SHOOTER

A new student problem has been submitted.

Student Name:
{name}

Enrollment Number:
{enrollment}

Problem:
{description}

Please login to the Mentor Dashboard to provide a solution.
"""

    message = MIMEText(body)

    message["Subject"] = subject
    message["From"] = SENDER_EMAIL
    message["To"] = MENTOR_EMAIL

    try:

        server = smtplib.SMTP(
            "smtp.gmail.com",
            587
        )

        server.starttls()

        server.login(
            SENDER_EMAIL,
            GMAIL_APP_PASSWORD
        )

        server.sendmail(
            SENDER_EMAIL,
            MENTOR_EMAIL,
            message.as_string()
        )

        server.quit()

        return True

    except Exception as error:

        print("Email Error:", error)

        return False

@app.route("/submit", methods=["POST"])
def submit():

    name = request.form.get("name", "").strip()
    enrollment = request.form.get("enrollment", "").strip()
    description = request.form.get("description", "").strip()

    if not name or not enrollment or not description:

        return "All fields are required.", 400


    conn = sqlite3.connect(DATABASE)

    cursor = conn.cursor()

    cursor.execute("""
        INSERT INTO problems
        (name, enrollment, description)
        VALUES (?, ?, ?)
    """, (
        name,
        enrollment,
        description
    ))

    conn.commit()
    conn.close()


    # Email mentor
    send_email(
        name,
        enrollment,
        description
    )


    return render_template_string(
        SUCCESS_PAGE,
        enrollment=enrollment
    )

    if request.method == "POST":

        username = request.form.get(
            "username",
            ""
        )

        password = request.form.get(
            "password",
            ""
        )

            password == MENTOR_PASSWORD

            session["mentor_logged_in"] = True

            return redirect(
                url_for("mentor_dashboard")
            )

        else:

            error = "Invalid username or password."


    return render_template_string(
        LOGIN_PAGE,
        error=error
    )

# SAVE MENTOR SOLUTION
# =====================================================

@app.route(
    "/mentor/solve/<int:problem_id>",
    methods=["POST"]
)

@mentor_required
def solve(problem_id):

    solution = request.form.get(
        "solution",
        ""
    ).strip()


    if not solution:

        return "Solution cannot be empty.", 400


    conn = sqlite3.connect(
        DATABASE
    )

    cursor = conn.cursor()


    cursor.execute("""
        UPDATE problems

        SET solution = ?,
            status = 'Solved'

        WHERE id = ?
    """, (
        solution,
        problem_id
    ))


    conn.commit()

    conn.close()


    return redirect(
        url_for("mentor_dashboard")
    )
